38: class PluginManager(
39:     private val capabilityRegistry: CapabilityRegistry,
40:     private val commandRegistry: CommandRegistry,
41:     private val parentScope: CoroutineScope = CoroutineScope(SupervisorJob() + Dispatchers.IO),
42: ) {
43:     private val plugins = ConcurrentHashMap<String, RegisteredPlugin>()
44:     private val statusByPlugin = ConcurrentHashMap<String, PluginStatus>()
45:     private val logTag = "FR3K.plugin"
46: 
47:     val pluginList: List<Fr3kPlugin> get() = plugins.values.map { it.plugin }
48:     val statuses: Map<String, PluginStatus> get() = statusByPlugin.toMap()
49: 
50:     fun register(plugin: Fr3kPlugin) {
51:         if (plugins.containsKey(plugin.pluginId)) {
52:             Log.w(logTag, "plugin ${plugin.pluginId} already registered; replacing")
53:             unregister(plugin.pluginId)
54:         }
55:         plugins[plugin.pluginId] = RegisteredPlugin(plugin)
56:         statusByPlugin[plugin.pluginId] = PluginStatus.REGISTERED
57:     }
58: 
59:     fun start(pluginId: String) {
60:         val reg = plugins[pluginId] ?: run {
61:             Log.w(logTag, "no such plugin: $pluginId"); return
62:         }
63:         val plugin = reg.plugin
64:         reg.scope = CoroutineScope(SupervisorJob() + Dispatchers.IO + parentScope.coroutineContext)
65:         reg.startJob = reg.scope!!.launch {
66:             runCatching {
67:                 statusByPlugin[pluginId] = PluginStatus.STARTING
68:                 capabilityRegistry.registerAll(pluginId, plugin.capabilities())
69:                 plugin.commands().forEach { commandRegistry.register(it) }
70:                 plugin.start()
71:                 statusByPlugin[pluginId] = PluginStatus.RUNNING
72:             }.onFailure {
73:                 Log.e(logTag, "plugin $pluginId start failed", it)
74:                 statusByPlugin[pluginId] = PluginStatus.FAILED
75:                 capabilityRegistry.unregisterAllByOwner(pluginId)
76:                 commandRegistry.unregisterByPlugin(pluginId)
77:             }
78:         }
79:     }
80: 
81:     fun unregister(pluginId: String) {
82:         val reg = plugins.remove(pluginId) ?: return
83:         reg.startJob?.cancel()
84:         reg.scope?.cancel()
85:         capabilityRegistry.unregisterAllByOwner(pluginId)
86:         commandRegistry.unregisterByPlugin(pluginId)
87:         statusByPlugin[pluginId] = PluginStatus.UNREGISTERED
88:     }
89: 
90:     fun startAll() = plugins.keys.toList().forEach { start(it) }
